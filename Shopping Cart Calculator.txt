double price1 = Convert.ToDouble(Console.ReadLine());

Console.Write("Enter Product 1 quantity: ");
int quantity1 = Convert.ToInt32(Console.ReadLine());

Console.Write("Enter Product 2 price: ");
double price2 = Convert.ToDouble(Console.ReadLine());

Console.Write("Enter Product 2 quantity: ");
int quantity2 = Convert.ToInt32(Console.ReadLine());

Console.Write("Enter Product 3 price: ");
double price3 = Convert.ToDouble(Console.ReadLine());

Console.Write("Enter Product 3 quantity: ");
int quantity3 = Convert.ToInt32(Console.ReadLine());

Console.Write("Enter Discount rate (%):");
double discountRate = Convert.ToDouble(Console.ReadLine());

// Calculation 
double totalPrice = (price1 * quantity1) + (price2 * quantity2) + (price3 * quantity3);
double discountAmount = totalPrice * (discountRate / 100);
double total = totalPrice - discountAmount;

//Boolean expressions
bool is5000 = total >= 5000;
bool is10000 = total >= 10000;
bool discountGreaterThan1000 = discountRate > 1000;
